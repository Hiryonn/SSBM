Array.prototype.randomize = function() {
 var i = this.length;
 if (i == 0) return;
 while (--i) {
  var j = Math.floor(Math.random()*(i+1));
  var tmp1 = this[i];
  var tmp2 = this[j];
  this[i] = tmp2;
  this[j] = tmp1;
 }
 return this;
}
