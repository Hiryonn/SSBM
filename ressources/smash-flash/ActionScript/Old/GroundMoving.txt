	function checkhurt() {
		hurtdelay++;
		if (hurtdelay>=recoveryrate and hurt) {
			hurt = false;
			if (!onground) {
				jumpa();
			} else if (onground and speed !== 0) {
				running();
			} else {
				standing();
			}
		}
	}
	function checkhit() {
		if (!fading) {
			if (_parent.player1.hitTest(_parent.player2.attackbox)) {
				health += _parent.player2.damage;
				health = Math.round(health);
				hit();
				lasthit = 2;
			}
			if (_parent.player1.hitTest(_parent.player3.attackbox)) {
				health += _parent.player3.damage;
				health = Math.round(health);
				hit();
				lasthit = 3;
			}
			if (_parent.player1.hitTest(_parent.player4.attackbox)) {
				health += _parent.player4.damage;
				health = Math.round(health);
				hit();
				lasthit = 4;
			}
			if (_parent.player1.hitTest(_parent.p2attackbox)) {
				health += _parent.player2.damage;
				health = Math.round(health);
				hit();
				lasthit = 2;
			}
			if (_parent.player1.hitTest(_parent.p3attackbox)) {
				health += _parent.player3.damage;
				health = Math.round(health);
				hit();

	function hit() {
		if (!hurt) {
			hurtdelay = 0;
			attacking = false;
			vulnerable = false;
			ducking = false;
			control = false;
			rev = false;
			down = false;
			rolling = false;
			_parent.player1.gotoAndPlay("hurt");
			_parent.player1.punch.start();
			if (!onground) {
				if (speed == 0) {
					speed = -2;
				} else if (speed>0) {
					speed = -2;
				} else if (speed<0) {
					speed = 2;
				}
				jump = -1;
				speed *= (health/healthdivision)*knockback;
				jump *= (health/healthdivision)*knockup;
			} else if (onground) {
				if (speed == 0) {
					if (Math.random()>.5) {
						speed = -2;
					} else {
						speed = 2;
					}
				} else if (speed>0) {
					speed = -2;
				} else if (speed<0) {
					speed = 2;
				}
				rev = false;
				rolling = false;
				ducking = false;
				onground = false;
				jumping = true;
				jump = -1;
				speed *= (health/healthdivision)*knockback;
				jump *= (health/healthdivision)*knockup;
				this._y -= jump;
				_parent.platform1._y -= _parent.ground.jump;
				_parent.platform2._y -= _parent.ground.jump;
				_parent.platform3._y -= _parent.ground.jump;
				_parent.platform4._y -= _parent.ground.jump;
				_parent.platform5._y -= _parent.ground.jump;
				_parent.player2._y -= _parent.ground.jump;
				_parent.player3._y -= _parent.ground.jump;
				_parent.player4._y -= _parent.ground.jump;
			}
		}
		hurt = true;
	}

	function checkduck() {
		if (onground and !hurt and !rolling) {
			if (down and !ducking) {
				ducking = true;
				ducka();
			}
			if (!down and ducking and !control and !rolling and !rev) {
				ducking = false;
				stand();
			}
			if (ducking and control) {
				if (!rev) {
					rolla();
				}
				damage = revdamage;
				rev = true;
				sounddelay++;
				if (sounddelay>12) {
					_parent.player1.revving.start();
					sounddelay = 0;
				}
			}
			if (rev) {
				if (facing == "right") {
					speed = revspeed;
				} else if (facing == "left") {
					speed = -revspeed;
				}
				rolling = true;
				rev = false;
				ducking = false;
				_parent.player1.shoot.start();
			}
		}
	}
	function checkattack() {
		attackdelay++;
		if (!hurt) {
			if (attackdelay>attackspace and control and right and facing == "right" and !ducking and !rev and !rolling) {
				vulnerable = false;
				attacking = true;
				attackdelay = 0;
				_parent.player1.aforward.start();
				damage = smash;
				attack2();
			} else if (attackdelay>attackspace and control and right and facing == "left" and !ducking and !rev and !rolling) {
				vulnerable = false;
				attacking = true;
				attackdelay = 0;
				_parent.player1.aforward.start();
				damage = smash;
				attack2();
			} else if (attackdelay>attackspace and control and up and !ducking and !rev and !rolling) {
				vulnerable = false;
				attacking = true;
				attackdelay = 0;
				_parent.player1.aup.start();
				damage = air;
				attackup();
			} else if (attackdelay>attackspace and control and down and !onground and !ducking and !rev and !rolling) {
				vulnerable = false;
				attacking = true;
				attackdelay = 0;
				_parent.player1.adown.start();
				damage = air;
				attackdown();
			} else if (attackdelay>attackspace and control and !ducking and !rev and !rolling) {
				vulnerable = false;
				attacking = true;
				attackdelay = 0;
				_parent.player1.anormal.start();
				damage = normal;
				attack();
			}
			if (attacking) {
				if (facing == "right" and speed<10) {
					speed += attackboost;
				} else if (facing == "left" and speed>-10) {
					speed -= attackboost;
				}
			}
			if (attackdelay>attacklength) {
				vulnerable = true;
				attacking = false;
			}
			if (!onground and attackdelay == attacklength) {
				jumpa();
			}
			if (onground and attackdelay == attacklength) {
				attacking = false;
				standing();
			}
			if (onground and attackdelay>attacklength and attacking) {
				attacking = false;
				standing();
			}
			if (!onground and attackdelay>attacklength and attacking) {
				attacking = false;
			}
		}
	}
	function goright() {
		if (right && speed<maxmove and !ducking and !rev and !rolling and !hurt) {
			if (!movingright and !jumping) {
				if (facing !== "right") {
					_parent.player1._x = _parent.player1._x-_parent.player1._width;
					_parent.player1._xscale = Math.abs(_parent.player1._xscale);
				}
				facing = "right";
				if (speed<-10 and onground) {
					speed += turnaround;
					_parent.player1.turn.start();
				}
				if (vulnerable) {
					running();
				}
			} else if (!movingright and jumping) {
				if (facing !== "right") {
					_parent.player1._x = _parent.player1._x-_parent.player1._width;
					_parent.player1._xscale = Math.abs(_parent.player1._xscale);
				}
				facing = "right";
				if (speed<-10 and onground) {
					speed += 8;
					_parent.player1.turn.start();
				}
			}
			if (onground and vulnerable) {
				running();
			}
			this._x -= speed;
			_parent.platform1._x -= _parent.ground.speed;
			_parent.platform2._x -= _parent.ground.speed;
			_parent.platform3._x -= _parent.ground.speed;
			_parent.platform4._x -= _parent.ground.speed;
			_parent.platform5._x -= _parent.ground.speed;
			_parent.player2._x -= _parent.ground.speed;
			_parent.player3._x -= _parent.ground.speed;
			_parent.player4._x -= _parent.ground.speed;
			tempspeed = speed;
			speed += .5;
			movingright = true;
			movingleft = false;
		} else {
			movingright = false;
		}
	}
	function goleft() {
		if (left && speed>-maxmove and !ducking and !rev and !rolling and !hurt) {
			if (!movingleft and !jumping) {
				if (facing !== "left") {
					_parent.player1._x = _parent.player1._x+_parent.player1._width;
					_parent.player1._xscale = -Math.abs(_parent.player1._xscale);
				}
				facing = "left";
				if (vulnerable) {
					running();
				}
				if (speed>10 and onground) {
					speed -= turnaround;
					_parent.player1.turn.start();
				}
			} else if (!movingleft and jumping) {
				if (facing !== "left") {
					_parent.player1._x = _parent.player1._x+_parent.player1._width;

				if (speed>10 and onground) {
					speed -= 8;

				}
				facing = "left";
			}
			if (onground) {
				if (vulnerable) {
					running();
				}
			}
			this._x -= speed;
			_parent.platform1._x -= _parent.ground.speed;
			_parent.platform2._x -= _parent.ground.speed;
			_parent.platform3._x -= _parent.ground.speed;
			_parent.platform4._x -= _parent.ground.speed;

			onground = true;
			jumping = false;
			jump = 0;
			lasthit = 0;
		} else if (_parent.platform5.hitTest(_parent.player1.feet) and falling and _parent.player1._y+35<_parent.platform5._y and !onground and !down) {
			if (jumping and !hurt) {
				standing();
			}
			if (hurtdelay>=recoveryrate and hurt) {
				hurt = false;
			}
			onground = true;
			jumping = false;
			jump = 0;
			lasthit = 0;
		} else if (!_parent.ground.hitTest(_parent.player1.feet) and !_parent.platform1.hitTest(_parent.player1.feet) and !_parent.platform1.hitTest(_parent.player1.feet) and !_parent.platform2.hitTest(_parent.player1.feet) and !_parent.platform3.hitTest(_parent.player1.feet) and !_parent.platform4.hitTest(_parent.player1.feet) and !_parent.platform5.hitTest(_parent.player1.feet)) {
			onground = false;
			jumping = true;
		}
	}
	function checkspeed() {
		if (speed>-1 and speed<1 and !movingleft and !movingright) {
			rolling = false;
			speed = 0;
		}
	}
	function checkifmoving() {
		if (speed == 0 and !jumping and !movingright and !movingleft and !rolling) {
			if (!attacking and !ducking and !rev and !rolling and _parent.player1._currentframe !== 1) {
				standing();
			}
			moving = false;
		}
	}
}
