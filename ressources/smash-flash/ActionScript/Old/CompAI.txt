		if (_parent.player1._x>this._x and speed<maxmove and (Math.sqrt(Math.pow((this._x-_parent.player1._x), 2)+Math.pow(this._y-_parent.player1._y, 2)))>attackrange and !hurt and !rolling and !cliff and temprun<followchance) {
			if (!movingright and !jumping) {
				if (facing !== "right") {
					this._x = this._x-this._width;
					this._xscale = Math.abs(this._xscale);
				}
				facing = "right";
				if (speed<-10 and onground) {
					speed += 8;
					_parent.player2.turn.start();
				}
				running();
			} else if (!movingright and jumping) {
				if (facing !== "right") {
					this._x = this._x-this._width;
					this._xscale = Math.abs(this._xscale);
				}
				facing = "right";
			}
			if (onground) {
				running();
			}
			this._x += speed;
			tempspeed = speed;
			speed += .5;
			movingright = true;
			movingleft = false;
		} else {
			movingright = false;
		}
	}
	function goleft() {
		temprun = Math.random();
		if (_parent.player1._x<this._x and (Math.sqrt(Math.pow((this._x-_parent.player1._x), 2)+Math.pow(this._y-_parent.player1._y, 2)))>attackrange and speed>-maxmove and !hurt and !rolling and !cliff and temprun<followchance) {
			if (!movingleft and !jumping) {
				if (facing !== "left") {
					this._x = this._x+this._width;
					this._xscale = -Math.abs(this._xscale);
				}
				if (speed>10 and onground) {
					speed -= 8;
					_parent.player2.turn.start();

				facing = "left";

					this._x = this._x+this._width;

				}
				facing = "left";
			}

			tempspeed = speed;
			speed -= .5;
			movingleft = true;
			movingright = false;
		} else {
			movingleft = false;
		}
	}
	function player2slow() {
		if (!movingright and !movingleft and speed>0) {
			this._x += speed;
			tempspeed = speed;
			speed -= 1;
		}
		if (!movingright and !movingleft and speed<0) {
			this._x += speed;
			tempspeed = speed;
			speed += 1;
		}
	}
	function player2jump() {
		if (_parent.player1._y<this._y-this._height-20 and onground and (Math.sqrt(Math.pow((this._x-_parent.player1._x))), 2)<jumprange and !hurt and Math.random()<jumpchance) {
			if (onground) {
				rev = false;
				rolling = false;
				ducking = false;
				_parent.player2.jumpsound.start();
				jumpa();
			}
			falling = false;
			jumping = true;
			jump = maxjump;
			this._y += jump;
			damage = jumpdamage;
			onground = false;
		} else if (jumping and !onground) {
			this._y += jump;
			jump++;
		}
		if (!falling and jump>=0) {
			falling = true;
			fall();
		}
	}

			rolling = false;
